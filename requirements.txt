# --- LLM transport ---

python-dotenv==1.2.4

# --- Speech-to-text ---
# sounddevice bundles PortAudio, so PyAudio is NOT needed. This matters:
# PyAudio has no prebuilt wheel for Python 3.14 and requires a C toolchain.
SpeechRecognition==3.17.0
sounddevice==0.5.6
numpy==2.5.4

# Butang Q untuk pause/resume dalam audio/stt.py

# --- Text-to-speech ---
# edge-tts gives free Indonesian neural voices (id-ID-ArdiNeural /
# id-ID-GadisNeural) with no API key and no ffmpeg.
edge-tts==7.3.1

# wheels for Python 3.14 whereas pygame itself does not. Also what
# "Voice Visualization" needs.
pygame-ce==2.5.8

# Pemain audio ganti, kalau pygame buat masalah (lihat backup/).
playsound3==3.3.2

# Offline fallback voice (Windows SAPI / macOS NSSpeech / espeak).
pyttsx3==2.99
